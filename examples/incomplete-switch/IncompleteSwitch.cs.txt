public record Text(string Value);
public record Missing;
public record RichText(string Html); // newly added case
public union ExampleDescription(Text, Missing, RichText);

public static class Example
{
    public static string Render(ExampleDescription description) => description switch
    {
        Text text => text.Value,
        Missing => "none",
        null => throw new InvalidOperationException("Uninitialized union.")
        // Missing RichText arm: non-exhaustive switch diagnostic.
        // Add: RichText text => text.Html,
    };
}
